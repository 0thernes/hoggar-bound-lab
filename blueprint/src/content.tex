\section{Foundations and Spectral Floor}

\begin{definition}[Pauli Operators and Symplectic Form]
\label{def:pauli}
\lean{HoggarBound.pauli, HoggarBound.symp, HoggarBound.lineSign, HoggarBound.flags, HoggarBound.violatedLines}
\leanok
For $N$ qubits ($D = 2^N$), each non-identity Pauli observable $P_u$ ($u \in \mathbb{F}_2^{2N} \setminus \{0\}$)
is a Hermitian unitary with commutation governed by the symplectic form $\langle u, v \rangle \in \mathbb{F}_2$
and commuting triple sign $\tau(u,v) \in \{\pm 1\}$.
\end{definition}

\begin{lemma}[Okumura Third-Moment Inequality]
\label{lem:okumura}
\lean{HoggarBound.sum_cube_le, HoggarBound.trace_cube_le}
\leanok
For any traceless Hermitian operator $A$ on $\mathbb{C}^D$ ($D \ge 2$) with $\operatorname{tr}(A^2) = D m$,
$(D-1)\operatorname{tr}(A^3) \le D(D-2)m\sqrt{(D-1)m}$.
\end{lemma}

\begin{theorem}[Theorem 1: Universal Spectral Floor]
\label{thm:spectral-floor}
\uses{def:pauli, lem:okumura}
\lean{HoggarBound.hoggar_bound, HoggarBound.NonVacuity.spectral_floor_nonvacuous}
\leanok
For every $N \ge 1$, every subset $S$ of non-identity $N$-qubit Pauli observables, and every $\pm 1$ assignment $f$,
$(D-1)\beta_6(S,f) \le (D-2)|S|\sqrt{(D-1)|S|}$.
\end{theorem}

\begin{corollary}[Corollary A: Asymptotic 1/2 Violated Fraction]
\label{cor:asymptotic-half}
\uses{thm:spectral-floor}
\lean{HoggarBound.AuditUpgrades.full_space_fraction_tendsto_half, HoggarBound.AuditUpgrades.hyperbolic_fraction_tendsto_half, HoggarBound.AuditUpgrades.elliptic_violated_fraction_eventually_gt}
\leanok
As $N \to \infty$, the minimum violated context fraction on $W(2N-1,2)$, $Q^+(2N-1,2)$, and $Q^-(2N-1,2)$
converges to $1/2$ (`Filter.Tendsto`).
\end{corollary}

\begin{theorem}[Theorem 2: Quaternionic / Kramers Spectral Floor]
\label{thm:kramers}
\uses{def:pauli, lem:okumura}
\lean{HoggarBound.kramers_even_mult, HoggarBound.hoggar_bound_quaternionic, HoggarBound.elliptic_fraction, HoggarBound.three_qubit_elliptic_contexts, HoggarBound.four_qubit_elliptic_contexts, HoggarBound.five_qubit_elliptic_contexts, HoggarBound.NonVacuity.kramers_nonvacuous}
\leanok

multiplicity, halving the effective dimension to $2^{N-1}$ and yielding $d(Q^-(5,2)) \ge 9$, $d(Q^-(7,2)) \ge 291$,
and $d(Q^-(9,2)) \ge 6501$.
\end{theorem}
